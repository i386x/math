% Text font size switches:
%   - helpers
\def\tenTextCommaSize{fivev}%
\def\nineTextCommaSize{fourvi}%
\def\sevenTextCommaSize{four}%
\def\sixTextCommaSize{threeiv}%
\def\MakeFontSwitch #1#2#3#4{%
  \namedef{#2}{%
    \def\tieaccent{\applyw{#1#3}\accent#4 }%
    \def\textcommafont{%
      \baselineskip\ZeroSkip%
      \applyw{\applyw{#1TextCommaSize}#2}%
    }%
    \applyw{#1#2}%
  }%
}%
\def\MakeFontSwitches #1{%
  \MakeFontSwitch{#1}{rm}{tsr}{26}%
  \MakeFontSwitch{#1}{sf}{i}{127}%
  \MakeFontSwitch{#1}{tt}{i}{127}%
  \MakeFontSwitch{#1}{it}{tsi}{26}%
  \MakeFontSwitch{#1}{sl}{tsi}{26}%
  \MakeFontSwitch{#1}{sc}{tsr}{26}%
  \MakeFontSwitch{#1}{bf}{tsb}{26}%
}%
\def\TextSizeSwitchBase #1{%
  \applyw{#1baseline}%
  \makestrutbox%
  \MakeFontSwitches{#1}%
}%
%   - normal size
\def\normal{%
  \TextSizeSwitchBase{ten}%
  \normalmath%
  \rm%
}%
%   - small size
\def\small{%
  \TextSizeSwitchBase{nine}%
  \smallmath%
  \rm%
}%
%   - script size (for footnote super scripts in text)
\def\scriptsize{%
  \TextSizeSwitchBase{seven}%
  \scriptsizemath%
  \rm%
}%
%   - small script size (for footnote super scripts in footnotes)
\def\smallscriptsize{%
  \TextSizeSwitchBase{six}%
  \smallscriptsizemath%
  \rm%
}%
