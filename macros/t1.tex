% Accents:
\def\`#1{%
  \ifx A#1^^c0\else%
  \ifx E#1^^c8\else%
  \ifx I#1^^cc\else%
  \ifx O#1^^d2\else%
  \ifx U#1^^d9\else%
  \ifx a#1^^e0\else%
  \ifx e#1^^e8\else%
  \ifx i#1^^ec\else%
  \ifx\i#1^^ec\else%
  \ifx o#1^^f2\else%
  \ifx u#1^^f9\else%
  {\accent0 #1}%
  \fi\fi\fi\fi\fi\fi\fi\fi\fi\fi\fi%
}%
\def\'#1{%
  \ifx C#1^^82\else%
  \ifx L#1^^88\else%
  \ifx N#1^^8b\else%
  \ifx R#1^^8f\else%
  \ifx S#1^^91\else%
  \ifx Z#1^^99\else%
  \ifx c#1^^a2\else%
  \ifx l#1^^a8\else%
  \ifx n#1^^ab\else%
  \ifx r#1^^af\else%
  \ifx s#1^^b1\else%
  \ifx z#1^^b9\else%
  \ifx A#1^^c1\else%
  \ifx E#1^^c9\else%
  \ifx I#1^^cd\else%
  \ifx O#1^^d3\else%
  \ifx U#1^^da\else%
  \ifx Y#1^^dd\else%
  \ifx a#1^^e1\else%
  \ifx e#1^^e9\else%
  \ifx i#1^^ed\else%
  \ifx\i#1^^ed\else%
  \ifx o#1^^f3\else%
  \ifx u#1^^fa\else%
  \ifx y#1^^fd\else%
  {\accent1 #1}%
  \fi\fi\fi\fi\fi\fi\fi\fi\fi\fi\fi\fi\fi\fi\fi\fi\fi\fi\fi\fi\fi\fi\fi\fi\fi%
}%
\def\^#1{%
  \ifx A#1^^c2\else%

  \ifx i#1^^ee\else%
  \ifx\i#1^^ee\else%
  \ifx o#1^^f4\else%
  \ifx u#1^^fb\else%
  {\accent2 #1}%
  \fi\fi\fi\fi\fi\fi\fi\fi\fi\fi\fi%
}%
\def\~#1{%
  \ifx A#1^^c3\else%

  \ifx n#1^^f1\else%
  \ifx o#1^^f5\else%
  {\accent3 #1}%
  \fi\fi\fi\fi\fi\fi%
}%

  \ifx E#1^^cb\else%
  \ifx I#1^^cf\else%
  \ifx O#1^^d6\else%
  \ifx U#1^^dc\else%

  \ifx\i#1i\else%

  \fi\fi\fi\fi\fi\fi\fi%

\def\b #1{%
  \leavevmode%
  \bgroup%
    \o@lign{%
      \relax#1\crcr%
      \hidewidth%
      \SlantedShift{-3ex}%
      \vbox to .2ex{\hbox{\char9}\vss}%
      \hidewidth%
    }%
  \egroup%
}%
\def\c #1{%
  \ifx S#1^^93\else%
  \ifx T#1^^95\else%
  \ifx s#1^^b3\else%
  \ifx t#1^^b5\else%
  \ifx C#1^^c7\else%
  \ifx c#1^^e7\else%
  \ifx g#1\textcommaabove{g}\else%
  \ifx G#1\textcommabelow{G}\else%
  \ifx K#1\textcommabelow{K}\else%
  \ifx k#1\textcommabelow{k}\else%
  \ifx L#1\textcommabelow{L}\else%
  \ifx l#1\textcommabelow{l}\else%
  \ifx N#1\textcommabelow{N}\else%
  \ifx n#1\textcommabelow{n}\else%
  \ifx R#1\textcommabelow{R}\else%
  \ifx r#1\textcommabelow{r}\else%
    \leavevmode%
    \setbox\Zero \hbox{#1}%
    \ifdim\ht\Zero=1ex%
      \accent11 #1%
    \else{%
      \ooalign{%
        \unhbox\Zero \crcr%
        \hidewidth\char11\hidewidth%
      }%
    }\fi%
  \fi\fi\fi\fi\fi\fi\fi\fi\fi\fi\fi\fi\fi\fi\fi\fi%
}%
\def\d #1{%
  \leavevmode%
  \bgroup%
    \o@lign{%
      \relax#1\crcr%
      \hidewidth\SlantedShift{-1ex}.\hidewidth%
    }%
  \egroup%

\def\k #1{%
  \ifx A#1^^81\else%
  \ifx E#1^^86\else%
  \ifx a#1^^a1\else%
  \ifx e#1^^a6\else%
  \ifx o#1\textogonekcentered{o}\else%
  \ifx O#1\textogonekcentered{O}\else%
    \leavevmode%
    \bgroup%
      \ooalign{%
        \null#1\crcr%
        \hidewidth\char12%
      }%
    \egroup%
  \fi\fi\fi\fi\fi\fi%
}%

  \leavevmode%
  \bgroup%
    \ooalign{%
      \null#1\crcr%

    }%

\def\Hwithstroke{%
  \leavevmode%
  \bgroup%
    \vphantom{H}%
    \textbox\Zero{H}%
    \ooalign{%
      H\cr%
      \hidewidth%
      \dimena .7\ht\Zero%
      \advance \dimena .1ex%
      \vrule height \dimena depth -.7\ht\Zero width .8\wd\Zero%
      \hidewidth\cr%
    }%
  \egroup%
}%
\def\hwithstroke{%
  \leavevmode%
  \bgroup%
    \vphantom{h}%
    \textbox\Zero{h}%
    \ooalign{%
      h\cr%
      \kern .075\wd\Zero%
      \dimena .7\ht\Zero%
      \advance \dimena .1ex%
      \vrule height \dimena depth -.7\ht\Zero width .4\wd\Zero%
      \hidewidth\cr%
    }%
  \egroup%
}%
