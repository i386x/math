% Stripping:
\def\strip #1{\futurelet\@tempa\@strip@left #1\relax}%
\def\@strip@left{%
  \ifx\@tempa\spacetok%
    \let\next\@strip@left@cont%
  \else\ifx\@tempa\relax%
    \def\next:##1\relax##2{\let##2\empty}%
  \else%
    \let\next\@strip@right%
  \fi\fi%
  \next:%
}%
\def\@strip@left@cont: {\futurelet\@tempa\@strip@left}%
\def\@strip@right:#1\relax{\@strip@right@cont#1\\ \\\relax}%
\begingroup%
  \catcode`\^^@=12%
  \gdef\@strip@right@cont #1 \\#2\relax{%
    \if^^@#2^^@%
      \let\next\@strip@done%
    \else%
      \let\next\@strip@right@cont%
    \fi%
    \next#1\\ \\\relax%
  }%
\endgroup%
\def\@strip@done #1\\#2\relax#3{\def#3{#1}}%

% Parsing:
\def\ParseSubject #1#2{\applyw{@parse@subject@#2}#1}%
\namedef{@parse@subject@.}#1#2.{%
  \futurelet\@tempa\@parse@subject@cont #2 ()\relax#1%
}%
\namedef{@parse@subject@:}#1#2:{%
  \futurelet\@tempa\@parse@subject@cont #2 ()\relax#1%
}%
\def\@parse@subject@cont{%
  \ifx\@tempa\asterisktok%
    \afterfi\@parse@subject@final@a%
  \else%
    \afterfi\@parse@subject@final@b%
  \fi%
}%
\def\@parse@subject@final@a*#1 (#2)#3\relax#4{#4{*}{#1}{#2}}%
\def\@parse@subject@final@b #1 (#2)#3\relax#4{#4{}{#1}{#2}}%
\def\ParseCounterSpec #1#2{\@parse@counter@spec@try@eq #2=\relax#1}%
\def\@parse@counter@spec@try@eq #1=#2\relax{%
  \if$#2$%
    \afterfi{\@parse@counter@spec@try@slash #1/\relax{0}}%
  \else%
    \afterfi{\@parse@counter@spec@eat@eq #1=#2\relax}%
  \fi%
}%
\def\@parse@counter@spec@eat@eq #1=#2=\relax{%
  \@parse@counter@spec@try@slash #1/\relax{#2}%
}%
\def\@parse@counter@spec@try@slash #1/#2\relax{%
  \if$#2$%
    \afterfi{\@parse@counter@spec@try@paren #1(\relax{1}}%
  \else%
    \afterfi{\@parse@counter@spec@eat@slash #1/#2\relax}%
  \fi%
}%
\def\@parse@counter@spec@eat@slash #1/#2/\relax{%
  \@parse@counter@spec@try@paren #1(\relax{#2}%
}%
\def\@parse@counter@spec@try@paren #1(#2\relax{%
  \if$#2$%
    \afterfi{\@parse@counter@spec@final{#1}{}}%
  \else%
    \afterfi{\@parse@counter@spec@eat@paren #1(#2\relax}%
  \fi%
}%
\def\@parse@counter@spec@eat@paren #1(#2)(\relax{%
  \@parse@counter@spec@final{#1}{#2}%
}%
\def\@parse@counter@spec@final #1#2#3#4#5{#5{#1}{#2}{#3}{#4}}%

% Character category utilities:
\def\changecat #1#2{%
  \namelet{@restore@\number`#1@catcode}\empty%
  \ifnum#2=\catcode`#1\relax\else%
    \nameedef{@restore@\number`#1@catcode}{%
      \catcode`\noexpand#1=\the\catcode`#1\relax%
    }%
    \catcode`#1=#2\relax%
  \fi%
}%
\def\restorecat #1{\applyw{@restore@\number`#1@catcode}}%
\def\makeatletter{\changecat\@\CatLetter}%
\def\makeatother{\changecat\@\CatOther}%

% Registers utilities:
\def\ensurereg #1#2#3{%
  \applycw\ifx{#2}\relax%
    \applyww{new#1}{#2}%
  \fi%
  \applyw{#2}=#3\relax%
}%
\def\newcounter #1{%
  \ParseCounterSpec\@newcounter{#1}%
}%
\def\@newcounter #1#2#3#4{%
  \ensurereg{count}{c@#1}{#4}%
  \nameedef{c@#1@style}{#3}%
  \namedef{c@#1@hooks}{}%
  \if$#2$\else%
    \applycw\addtoks{c@#2@hooks}{%
      \applyw{c@#1@hooks}%
      \global\applyw{c@#1}=#4\relax%
    }%
  \fi%
}%
\def\stepcounter #1{%
  \global\applycw\advance{c@#1}\One%
  \applyw{c@#1@hooks}%
}%

}%

  \next%
