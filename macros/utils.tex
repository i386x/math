% Sorting:
\newif\ifle \lefalse%
\def\mergesort #1#2{%
  % #1: a comparison macro
  % #2: a macro containing a sequence of control sequences
  \begingroup%
    \@mergesort@init #1#2%
    \@mergesort@run #1#2%
    \ifnum\counta<\One%
      \wraning{\string\mergesort\space is broken.}%
    \fi%
    \throwover #2%
  \endgroup%
}%
\def\@mergesort@init #1#2{%
  % #1, #2 as in \mergesort
  \counta \Zero%
  \unwrapttt\@merge@singletons#1#2#2\sentinel\sentinel%
}%
\def\@merge@singletons #1#2#3#4{%
  % #1: a comparison macro
  % #2: a container
  % #3, #4: elements
  \ifx#3\sentinel%
    % We reached the end, align to the even number of groups
    \@mergesort@align@groups #2%
  \else\ifx#4\sentinel%
    % One singleton group
    \addtoks #2{{#3}}%
    \advance \counta \One%
    \@mergesort@align@groups #2%
  \else%
    % Two elements
    #1#3#4%
    \ifle%
      % #3 <= #4
      \addtoks #2{{#3#4}}%
    \else%
      % #3 > #4
      \addtoks #2{{#4#3}}%
    \fi%
    \advance \counta \One%
    \afterfifi{\@merge@singletons#1#2}%
  \fi\fi%
}%
\def\@mergesort@align@groups #1{%
  \ifodd\counta%
    \addtoks #1{{}}%
    \advance \counta \One%
  \else%
    % Add two empty groups to avoid groups merging crash
    \addtoks #1{{}{}}%
    \advance \counta \Two%
  \fi%
}%
\def\@mergesort@run #1#2{%
  \ifnum\counta<\Two\else%
    \unwrapttt\@merge@groups#1#2#2\sentinel%
    % Sanitize \@merge@groups result
    \ifnum\counta<\One%
      \def#2{{}}%
      \counta \One%
    \else\ifnum\counta=\One%
      % #2 is sorted
    \else%
      \@mergesort@align@groups #2%
    \fi\fi%
    \afterfi{\@mergesort@run #1#2}%
  \fi%
}%
\def\@merge@groups #1#2#3#4{%
  % Two groups are eaten
  \advance \counta -\Two%
  \def\@tempa{#3}%
  \def\@tempb{#4}%
  \ifx\@tempa\empty%
    % End reached => stop
    \def\next ##1\sentinel{}%
  \else\ifx\@tempb\empty%
    % Merge #3 with the empty group and stop
    \addtoks #2{{#3}}%
    \advance \counta \One%
    \def\next ##1\sentinel{}%
  \else%
    % Merge #3 and #4 together
    \let\@tempe\empty%
    \def\next{%
      % \@tempc := car(\@tempa)
      \carx\@tempc\@tempa%
      % \@tempd := car(\@tempb)
      \carx\@tempd\@tempb%
      % Compare \@tempc and \@tempd
      \ifx\@tempc\empty%
        \ifx\@tempd\empty%
          % End of merging
          \let\next\relax%
        \else%
          \@merge@groups@move\@tempe\@tempd\@tempb%
        \fi%
      \else%
        \ifx\@tempd\empty%
          \@merge@groups@move\@tempe\@tempc\@tempa%
        \else%
          \applycxx#1\@tempc\@tempd%
          \ifle%
            % \@tempc <= \@tempd
            \@merge@groups@move\@tempe\@tempc\@tempa%
          \else%
            % \@tempc > \@tempd
            \@merge@groups@move\@tempe\@tempd\@tempb%
          \fi%
        \fi%
      \fi%
      \next%
    }%
    \next%
    \applycctx\addtoks#2{\expandafter{\@tempe}}%
    \advance \counta \One%
    \def\next{\@merge@groups #1#2}%
  \fi\fi%
  \next%
}%
\def\@merge@groups@move #1#2#3{%
  % append(#1, #2)
  \applycctx\addtoks#1{#2}%
  % #3 := cdr(#3)
  \cdrx#3#3%
}%

% Vertical list utilities:

\def\xremovelastskip #1{%
  % Remove \lastskip if \lastskip <> #1
  \skipa #1\relax%
  \advance \skipa -\lastskip%
  \ifdim\skipa=\ZeroUnit\else%
    \vskip -\lastskip%
  \fi%
}%
\def\@can@break@here{%
  % Test whether the last element of the recent contributions list is a
  % \penalty greater or equal 10000, possibly followed by one non-zero glue
  % element. Since the test is not perfect, use this macro with caution
  \skipa \ZeroSkip%
  \ifdim\lastskip=\ZeroUnit\else%
    \skipa \lastskip%
    \unskip%
  \fi%
  \ifnum\lastpenalty<\TenThousands%
    \@nobreakfalse%
  \else%
    \@nobreaktrue%
  \fi%
  \ifdim\skipa=\ZeroUnit\else%
    \vskip \skipa%
  \fi%
}%
\def\addpenalty #1{%
  \@can@break@here%
  \if@nobreak\else%
    \ifdim\lastskip=\ZeroUnit%
      \penalty #1\relax%
    \else%
      \xremovelastskip\ZeroUnit%
      % \skipa is now -\lastskip
      \penalty #1\relax%
      \vskip -\skipa%
    \fi%
  \fi%
}%
\def\adjustvspace #1{%
  % If #1 >= 0, the new vertical space will be max(\lastskip, #1).
  % If #1 < 0, the new vertical space will be max(0, \lastskip) + #1. This is
  % useful when we need to make a room for the incoming positive \vskip: the
  % incoming positive \vskip will either reset the negative \lastskip to 0 or
  % it will be canceled by the positive \lastskip
  \ifdim\lastskip=\ZeroUnit%
    \vskip #1\relax%
  \else%
    \skipb \lastskip%
    \xremovelastskip{#1}%
    % \skipa is now (#1 - \lastskip), make \skipa to be #1
    \advance \skipa \skipb%
    % If \lastskip = #1, do not contribute any space (the \xremovelastskip
    % command is a no-op in this case)
    \ifdim\skipa=\skipb\else%
      \ifdim\skipb<\skipa%
        \vskip \skipa%
      \else\ifdim\skipa<\ZeroUnit%
        \ifdim\skipb>\ZeroUnit%
          \advance \skipa \skipb%
          \vskip \skipa%
        \fi%
      \fi\fi%
    \fi%
  \fi%
}%
