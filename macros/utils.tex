\def\@do@spaces{%
  % Spaces are ordered using T1 scheme, except en-dash and em-dash which are
  % placed after slash because they optically occupy more space:
  %   blank < visible < dash < slash < bar < back slash < en-dash < em-dash
  %   < underscore
  \do\ \do\textvisiblespace\do\slash\do\textbar\do\textbackslash\do\textendash%
  \do\textemdash\do\_\do\textunderscore%
}%

  %   # < $ < pound < % < per thousand < per ten thousand < & [OT1, complexity]

  \do\Orb\do\copyright\do\textasciicircum\do\textasciitilde%
  \do\#\do\$\do\textdollar\do\textsterling\do\%\do\textperthousand%
    \do\textpertenthousand\do\&%

\def\@prepare@symbol@sorting@preprocess #1{%
  \begingroup%
    % Protect text macros against expansion
    \@protect@text@macros%
    % Expand the rest
    \edef\@tempb{#1}%
    % Translate
    \let\@tempa\empty%
    \unwrapt\@translate@symbols\@tempb\sentinel\\%
    % Store translated result to #1
    \promoteas\@tempb#1%
  \endgroup%
}%

\def\@prepare@symbol@sorting@stage@i #1#2{%
  % Stage I:
  %   - normalize spaces
  %   - remove accents and punctuation
  %   - do not care about capitals
  \@protect@text@macros%
  \@@normalize@spaces%
  \@@remove@accents@and@punctuation%
  \@translate@modified@letters%
  % Do not care about capitals
  \tolower#1#1%
  \tolower#2#2%
}%
\def\@prepare@symbol@sorting@stage@ii #1#2{%
  % Stage II:
  %   - letters, symbols, and normalized spaces are same
  %   - normalize spaces
  %   - care about accents and modified letters
  %   - remove punctuation
  %   - do not care about capitals
  \@protect@text@macros%
  \@@normalize@spaces%
  \@@remove@punctuation%
  \@translate@modified@letters@ii%
  % Apply changes before processing accents
  \edef#1{#1}%
  \edef#2{#2}%
  % Process accents
  \def\do ##1{%
    % Define for every accent macro
    \def##1####1{%
      \applycw\ifx{\@codename####1}\relax%
        % A token with no code -> pass it without expansion
        \noexpand####1%
      \else%
        \ifnum\applyw{\@codename####1}>\applyw{\@codename\\}%
          \ifnum\applyw{\@codename####1}<\applyw{\@codename\@ml@f}%
            % A token is an accent command (accent applied on accent, which is
            % probably a mistake) -> consume it and go to the next token
            \afterfifi{##1}%
          \else%
            % A token is \@ml@f (accent applied on a modified letter, the codes
            % after \@ml@f do not appear in the token stream) -> emit accent
            % for the modified letter
            \applyw{\@codename\@ml@f\string##1}%
          \fi%
        \else%
          % A token is not an accent -> emit the accent macro code
          \applyw{\@codename##1}%
        \fi%
      \fi%
    }%
  }%
  \@do@accents%
  % Do not care about capitals
  \tolower#1#1%
  \tolower#2#2%
}%
\def\@prepare@symbol@sorting@stage@iii #1#2{%
  % Stage III:
  %   - letters (even accented and modified), symbols, and normalized spaces
  %     are same

  \@@normalize@spaces%
  \@@remove@accents@and@punctuation%
  \@translate@modified@letters%

  \edef#2{#2}%
}%

\def\comparetext #1#2{%
  \begingroup%
    \@prepare@symbol@sorting@preprocess #1%
    \@prepare@symbol@sorting@preprocess #2%
    \compare@text{i}#1#2%
    \ifnum\counta=\Zero\compare@text{ii}#1#2\fi%
    \ifnum\counta=\Zero\compare@text{iii}#1#2\fi%

  \endgroup%

}%

\def\@compare@symbols #1#2{%
  \counta \Zero%

\def\@compare@symbols@cont #1#2\sentinel#3#4\sentinel{%
  \let\next\relax%
  \if&#1#3&%
    % Both empty => left == right
  \else\if&#1&%
    % Left empty, right not => left < right
    \counta \MinusOne%
  \else\if&#3&%
    % Right empty, left not => left > right
    \counta \One%
  \else%
    \@compare@symbols@getcode\counta#1%
    \@compare@symbols@getcode\countb#2%
    \advance \counta -\countb%
    \ifnum\counta=\Zero%
      \def\next{\@compare@symbols@cont #2\sentinel#4\sentinel}%
    \fi%
  \fi\fi\fi%
  \next%
}%

%   - sorting
\def\mergesort #1#2{%
  % #1: a comparison macro
  % #2: a macro containing a sequence of control sequences
  \begingroup%
    \@mergesort@init #1#2%
    \@mergesort@run #1#2%
    \ifnum\counta<\One%
      \warning{\string\mergesort\space is broken.}%
    \fi%
    \promote #2%
  \endgroup%
}%
\def\@mergesort@init #1#2{%
  % #1, #2 as in \mergesort
  \counta \Zero%
  \unwrapttt\@merge@singletons#1#2#2\sentinel\sentinel%
}%
\def\@merge@singletons #1#2#3#4{%
  % #1: a comparison macro
  % #2: a container
  % #3, #4: elements
  \ifx#3\sentinel%
    % We reached the end, align to the even number of groups
    \@mergesort@align@groups #2%
  \else\ifx#4\sentinel%
    % One singleton group
    \addtoks #2{{#3}}%
    \advance \counta \One%
    \@mergesort@align@groups #2%
  \else%
    % Two elements
    #1#3#4%
    \ifle%
      % #3 <= #4
      \addtoks #2{{#3#4}}%
    \else%
      % #3 > #4
      \addtoks #2{{#4#3}}%
    \fi%
    \advance \counta \One%
    \afterfifi{\@merge@singletons#1#2}%
  \fi\fi%
}%
\def\@mergesort@align@groups #1{%
  \ifodd\counta%
    \addtoks #1{{}}%
    \advance \counta \One%
  \else%
    % Add two empty groups to avoid groups merging crash
    \addtoks #1{{}{}}%
    \advance \counta \Two%
  \fi%
}%
\def\@mergesort@run #1#2{%
  \ifnum\counta<\Two\else%
    \unwrapttt\@merge@groups#1#2#2\sentinel%
    % Sanitize \@merge@groups result
    \ifnum\counta<\One%
      \def#2{{}}%
      \counta \One%
    \else\ifnum\counta=\One%
      % #2 is sorted
    \else%
      \@mergesort@align@groups #2%
    \fi\fi%
    \afterfi{\@mergesort@run #1#2}%
  \fi%
}%
\def\@merge@groups #1#2#3#4{%
  % Two groups are eaten
  \advance \counta -\Two%
  \def\@tempa{#3}%
  \def\@tempb{#4}%
  \ifx\@tempa\empty%
    % End reached => stop
    \def\next ##1\sentinel{}%
  \else\ifx\@tempb\empty%
    % Merge #3 with the empty group and stop
    \addtoks #2{{#3}}%
    \advance \counta \One%
    \def\next ##1\sentinel{}%
  \else%
    % Merge #3 and #4 together
    \let\@tempe\empty%
    \def\next{%
      % \@tempc := car(\@tempa)
      \carx\@tempc\@tempa%
      % \@tempd := car(\@tempb)
      \carx\@tempd\@tempb%
      % Compare \@tempc and \@tempd
      \ifx\@tempc\empty%
        \ifx\@tempd\empty%
          % End of merging
          \let\next\relax%
        \else%
          \@merge@groups@move\@tempe\@tempd\@tempb%
        \fi%
      \else%
        \ifx\@tempd\empty%
          \@merge@groups@move\@tempe\@tempc\@tempa%
        \else%
          \applycxx#1\@tempc\@tempd%
          \ifle%
            % \@tempc <= \@tempd
            \@merge@groups@move\@tempe\@tempc\@tempa%
          \else%
            % \@tempc > \@tempd
            \@merge@groups@move\@tempe\@tempd\@tempb%
          \fi%
        \fi%
      \fi%
      \next%
    }%
    \next%
    \applycctx\addtoks#2{\expandafter{\@tempe}}%
    \advance \counta \One%
    \def\next{\@merge@groups #1#2}%
  \fi\fi%
  \next%
}%
\def\@merge@groups@move #1#2#3{%
  % append(#1, #2)
  \applycctx\addtoks#1{#2}%
  % #3 := cdr(#3)
  \cdrx#3#3%
}%

\def\@can@break@here{%
  % Test whether the last element of the recent contributions list is a
  % \penalty greater or equal 10000, possibly followed by one non-zero glue
  % element. Since the test is not perfect, use this macro with caution
  \skipa \ZeroSkip%
  \ifdim\lastskip=\ZeroUnit\else%

    \@nobreakfalse%

  \fi%

  \fi%

\def\adjustvspace #1{%
  % If #1 >= 0, the new vertical space will be max(\lastskip, #1).
  % If #1 < 0, the new vertical space will be max(0, \lastskip) + #1. This is
  % useful when we need to make a room for the incoming positive \vskip: the
  % incoming positive \vskip will either reset the negative \lastskip to 0 or
  % it will be canceled by the positive \lastskip
  \ifdim\lastskip=\ZeroUnit%
    \vskip #1\relax%
  \else%
    \skipb \lastskip%
    \xremovelastskip{#1}%
    % \skipa is now (#1 - \lastskip), make \skipa to be #1
    \advance \skipa \skipb%
    % If \lastskip = #1, do not contribute any space (the \xremovelastskip
    % command is a no-op in this case)
    \ifdim\skipa=\skipb\else%
      \ifdim\skipb<\skipa%
        \vskip \skipa%
      \else\ifdim\skipa<\ZeroUnit%
        \ifdim\skipb>\ZeroUnit%
          \advance \skipa \skipb%
          \vskip \skipa%
        \fi%
      \fi\fi%
    \fi%
  \fi%
}%
