% Accents (OT1):
%   - \`
\def\`#1{%
  \ifx i#1\`\i\else%
  \ifx j#1\`\j\else%
  {\accent18 #1}%
  \fi\fi%
}%
%   - \'
\def\'#1{%
  \ifx i#1\'\i\else%
  \ifx j#1\'\j\else%
  {\accent19 #1}%
  \fi\fi%
}%
%   - \v
\def\v #1{%
  \ifx i#1\v\i\else%
  \ifx j#1\v\j\else%
  {\accent20 #1}%
  \fi\fi%
}%
%   - \u
\def\u #1{%
  \ifx i#1\u\i\else%
  \ifx j#1\u\j\else%
  {\accent21 #1}%
  \fi\fi%
}%
%   - \=
\def\=#1{%
  \ifx i#1\=\i\else%
  \ifx j#1\=\j\else%
  {\accent22 #1}%
  \fi\fi%
}%
%   - \r
\def\r #1{%
  \ifx i#1\r\i\else%
  \ifx j#1\r\j\else%
  \ifx A#1%
    \leavevmode%
    \setbox\Zero \hbox{!}%
    \dimena \ht\Zero%
    \advance \dimena -1ex%
    \rlap{\raise .67\dimena \hbox{\char23}}%
    A%
  \else%
    {\accent23 #1}%
  \fi\fi\fi%
}%
%   - \^
\def\^#1{%
  \ifx i#1\^\i\else%
  \ifx j#1\^\j\else%
  {\accent94 #1}%
  \fi\fi%
}%
%   - \.
\def\.#1{%
  \ifx i#1`\i\else%
  \ifx\i#1`\i\else%
  \ifx j#1`\j\else%
  \ifx\j#1`\j\else%
  {\accent95 #1}%
  \fi\fi\fi\fi%
}%
%   - \H
\def\H #1{%
  \ifx i#1\H\i\else%
  \ifx j#1\H\j\else%
  {\accent125 #1}%
  \fi\fi%
}%
%   - \~
\def\~#1{%
  \ifx i#1\~\i\else%
  \ifx j#1\~\j\else%
  {\accent126 #1}%
  \fi\fi%
}%
%   - \"
\def\"#1{%
  \ifx i#1\"\i\else%
  \ifx j#1\"\j\else%
  {\accent127 #1}%
  \fi\fi%
}%
%   - \b
\def\b #1{%
  \leavevmode%
  \bgroup%
    \o@lign{%
      \relax#1\crcr%
      \hidewidth%
      \SlantedShift{-3ex}%
      \vbox to .2ex{\hbox{\char22}\vss}%
      \hidewidth%
    }%
  \egroup%
}%
%   - \c
\def\c #1{%
  \ifx g#1\textcommaabove{g}\else%
    \leavevmode%
    \setbox\Zero \hbox{#1}%
    \ifdim\ht\Zero=1ex%
      \accent24 #1%
    \else{%
      \ooalign{%
        \unhbox\Zero\crcr%
        \hidewidth\char24\hidewidth%
      }%
    }\fi%
  \fi%
}%
%   - \d
\def\d #1{%
  \leavevmode%
  \bgroup%
    \o@lign{%
      \relax#1\crcr%
      \hidewidth\SlantedShift{-1ex}.\hidewidth%
    }%
  \egroup%
}%
%   - \t
\def\t #1{%
  \leavevmode%
  \bgroup%
    \edef\next{\the\font}%
    \tieaccent\next#1%
  \egroup%
}%
%   - comma above helper
\def\textcommaabove #1{%
  \leavevmode%
  \bgroup%
    \ooalign{%
      \hidewidth\raise .7ex \hbox{\textcommafont`}\hidewidth\crcr%
      \null#1\crcr%
    }%
  \egroup%
}%
%   - comma below helper
\def\textcommabelow #1{%
  \leavevmode%
  \bgroup%
    \ooalign{%
      \null#1\crcr%
      \hidewidth\raise -.31ex \hbox{\textcommafont,}\hidewidth%
    }%
  \egroup%
}%

\def\MakeFontSwitch #1#2#3#4#5{%
  \namedef{#2}{%
    \def\tieaccent{\applyw{#1ts#3}\accent26 }%
    \def\textcommafont{%
      \baselineskip \ZeroSkip%
      \applyw{five#2}%
    }%
    \def\textdollarfont{\applyw{#1#4}}%
    \def\textsterlingfont{\applyw{#1#5}}%
    \def\copyrightfont{\applyw{#1ts#3}}%
    \applyw{#1#2}%

\def\makepage{%
  \ifnum\outputpenalty>-\TwentyThousands\else%
    % The page break was chosen at \penalty -\@MM or lesser. This happens only
    % in these cases:
    %
    %   * \end has been invoked: the tail of \box255 contains
    %     \hbox to \hsize{}, possibly preceded by \baselineskip, and \vfill;
    %
    %   * \supereject has been invoked: the tail of \box255 contains only
    %     \vfill.
    %
    % We are now about to cut the tail and measure the \box255 without it to
    % see if there are some lines of text remaining or the recent page will
    % contain only floats
    \setbox\Cclv \vbox{%
      \boxmaxdepth \maxdepth%
      \unvbox\Cclv%
      \ifnum\outputpenalty<-\ThirtyThousands%
        % We do not use \penalty lesser than -30 000 and thus we know that we
        % can only get here when the page break was chosen at \penalty -2^30,
        % issued by \end
        %
        % Remove \vfill
        \unskip%
        % Remove \hbox to \hsize{}
        \setbox\Zero \lastbox%
        \ifvoid\Zero%
          \warning{%
            While shipping page \the\pageno\space (\the\realpageno), the%
            \space expected \string\hbox\space to \string\hline{}, inserted%
            \space by \string\end, was not found. This may refer to a bug in%
            \space \string\makepage\space macro but it could also refer to a%
            \space situation when a custom \string\penalty, lesser than%
            \space -\number\ThirtyThousands, has been issued.%
          }%
        \fi%
        % Remove possible \baselineskip inserted before \hbox to \hsize{}
        \unskip%
        % Append back removed \vfill
        \vfill%
      \fi%
    }%
  \fi%
  \dimen\Zero \ht\Cclv%
  \advance \dimen\Zero \dp\Cclv%
  \ifdim\dimen\Zero>.5\topskip%
    \makepageT%
  \else%
    \makepageF%
  \fi%
}%
