% Accents (OT1):
%   - \`
\def\`#1{%
  \ifx i#1\`\i\else%
  \ifx j#1\`\j\else%
  {\accent18 #1}%
  \fi\fi%
}%
%   - \'
\def\'#1{%
  \ifx i#1\'\i\else%
  \ifx j#1\'\j\else%
  {\accent19 #1}%
  \fi\fi%
}%
%   - \v
\def\v #1{%
  \ifx i#1\v\i\else%
  \ifx j#1\v\j\else%
  {\accent20 #1}%
  \fi\fi%
}%
%   - \u
\def\u #1{%
  \ifx i#1\u\i\else%
  \ifx j#1\u\j\else%
  {\accent21 #1}%
  \fi\fi%
}%
%   - \=
\def\=#1{%
  \ifx i#1\=\i\else%
  \ifx j#1\=\j\else%
  {\accent22 #1}%
  \fi\fi%
}%
%   - \r
\def\r #1{%
  \ifx i#1\r\i\else%
  \ifx j#1\r\j\else%
  \ifx A#1%
    \leavevmode%
    \setbox\Zero\hbox{!}%
    \dimena \ht\Zero%
    \advance\dimena -1ex%
    \rlap{\raise .67\dimena\hbox{\char23}}%
    A%
  \else%
    {\accent23 #1}%
  \fi\fi\fi%
}%
%   - \^
\def\^#1{%
  \ifx i#1\^\i\else%
  \ifx j#1\^\j\else%
  {\accent94 #1}%
  \fi\fi%
}%
%   - \.
\def\.#1{%
  \ifx i#1`\i\else%
  \ifx\i#1`\i\else%
  \ifx j#1`\j\else%
  \ifx\j#1`\j\else%
  {\accent95 #1}%
  \fi\fi\fi\fi%
}%
%   - \H
\def\H #1{%
  \ifx i#1\H\i\else%
  \ifx j#1\H\j\else%
  {\accent125 #1}%
  \fi\fi%
}%
%   - \~
\def\~#1{%
  \ifx i#1\~\i\else%
  \ifx j#1\~\j\else%
  {\accent126 #1}%
  \fi\fi%
}%
%   - \"
\def\"#1{%
  \ifx i#1\"\i\else%
  \ifx j#1\"\j\else%
  {\accent127 #1}%
  \fi\fi%
}%
%   - \b
\def\b #1{%
  \leavevmode%
  \bgroup%
    \o@lign{%
      \relax#1\crcr%
      \hidewidth%
      \SlantedShift{-3ex}%
      \vbox to .2ex{\hbox{\char22}\vss}%
      \hidewidth%
    }%
  \egroup%
}%
%   - \c
\def\c #1{%
  \ifx g#1\textcommaabove{g}\else%
    \leavevmode%
    \setbox\Zero\hbox{#1}%
    \ifdim\ht\Zero=1ex%
      \accent24 #1%
    \else{%
      \ooalign{%
        \unhbox\Zero\crcr%
        \hidewidth\char24\hidewidth%
      }%
    }\fi%
  \fi%
}%
%   - \d
\def\d #1{%
  \leavevmode%
  \bgroup%
    \o@lign{%
      \relax#1\crcr%
      \hidewidth\SlantedShift{-1ex}.\hidewidth%
    }%
  \egroup%
}%
%   - \t
\def\t #1{%
  \leavevmode%
  \bgroup%
    \edef\next{\the\font}%
    \tieaccent\next#1%
  \egroup%
}%
%   - comma above helper
\def\textcommaabove #1{%
  \leavevmode%
  \bgroup%
    \ooalign{%
      \hidewidth\raise .7ex\hbox{\textcommafont`}\hidewidth\crcr%
      \null#1\crcr%
    }%
  \egroup%
}%
%   - comma below helper
\def\textcommabelow #1{%
  \leavevmode%
  \bgroup%
    \ooalign{%
      \null#1\crcr%
      \hidewidth\raise -.31ex\hbox{\textcommafont,}\hidewidth%
    }%
  \egroup%
}%

\def\makestrutbox{%
  \setbox\strutbox\hbox{%
    \vrule width\ZeroUnit height.7\baselineskip depth.3\baselineskip%

\def\MakeFontSwitch #1#2#3#4#5{%
  \namedef{#2}{%
    \def\tieaccent{\applyw{#1ts#3}\accent26 }%
    \def\textcommafont{%
      \baselineskip\ZeroSkip%
      \applyw{five#2}%
    }%
    \def\textdollarfont{\applyw{#1#4}}%
    \def\textsterlingfont{\applyw{#1#5}}%
    \def\copyrightfont{\applyw{#1ts#3}}%
    \applyw{#1#2}%
  }%
}%

\def\footnote #1{%
  \global\advance\fncount\One%
  \xdef\thefncount{\the\fncount}%
  \leavevmode%
  \edef\@tempsf{\the\spacefactor}%
  \nobreak%
  \makefnmark%
  \spacefactor\@tempsf%
  \insert\footins{%
    \small%
    \interlinepenalty\interfootnotelinepenalty%
    \splittopskip\footnotesep%
    \splitmaxdepth\dp\strutbox%
    \floatingpenalty\TwentyThousands%
    \parboxrestore%
    \makefntext{%
      \vrule width\ZeroUnit height\footnotesep depth\ZeroUnit%
      \ignorespaces%
      #1%
      \unskip%
      \ifhmode\nobreak\fi%
      \vrule width\ZeroUnit height\ZeroUnit depth\dp\strutbox%
    }%
  }%
}%

  \let\@ragged@bottom\empty%
}%
\def\raggedbottom{%
  \def\@ragged@bottom{\vskip \ZeroUnit plus.0001fil\relax}%
}%
\def\supereject{%
  \par%
  \hbox to\hsize{}%
  \nobreak%
  \vfill%
  \penalty -\TwentyThousands%
}%
\def\normaloutput{%
  \shipout\vbox{%
    \offinterlineskip%
    \moveright\sidemargin\vbox{%
      \makeheadline%
      \makepage%
      \makefootline%
    }%
  }%
  \advancepageno%
  \ifnum\outputpenalty>-\TwentyThousands\else%
    \ifnum\insertpenalties>\Zero%
      \supereject%
    \fi%
  \fi%
}%
\def\makeheadline{%
  \setbox\Zero\vbox to\headheight{%
    \vfil%
    \hbox to\hsize{\@the@head}%
  }%
  \dp\Zero\ZeroUnit%
  \box\Zero%
  \vskip\headsep%
}%
\def\makepage{%
  \ifnum\outputpenalty>-\TwentyThousands\else%
    % What contains the tail of \box255 when we arrive here?
    % If we get here by \end, it means that the page break was chosen before
    % \hbox to \hsize{} (because after this box all elements, including the
    % \penalty -2^30, are discardable). Similarly to \supereject, where we also
    % forbid break between \hbox to \hsize{} and \penalty-\@MM so these are
    % always together (preferring \supereject over \end prevents
    % \hbox to \hsize{} to be the last element on a page).
    %
    % We are now about to cut the tail and measure the \box255 without it to
    % see if there are some lines of text remaining or the recent page will
    % contain only floats.
    \setbox\Cclv\vbox{%
      \counta 10% prevent infinite loop
      \boxmaxdepth\maxdepth%
      \unvbox\Cclv%
      \loop%
        \advance\counta\MinusOne%
        \unskip\unpenalty%
        \ifnum\counta>\Zero%
          \setbox\Zero\lastbox%
        \else%
          \setbox\Zero\hbox{}% this stops the loop
        \fi%
        \ifvoid\Zero%
      \repeat%
      % Remove \topskip or \baselineskip:
      \unskip%
      % Append removed \vfill:
      \vfill%
    }%
  \fi%
  \dimen\Zero\ht\Cclv%
  \advance\dimen\Zero\dp\Cclv%
  \ifdim\dimen\Zero>.5\topskip%
    \makepageT%
  \else%
    \makepageF%
  \fi%
}%
\def\makepageT{%
  \setbox\Zero\box\Cclv%
  \ifvoid\topins\else%
    \setbox\Zero\vbox{%
      \boxmaxdepth\maxdepth%
      \unvbox\topins%
      \unskip%
      \vskip\textfloatsep%
      \unvbox\Zero%
    }%
  \fi%
  \ifvoid\botins\else%
    \setbox\Zero\vbox{%
      \unvbox\Zero%
      \vskip\textfloatsep%
      \unvbox\botins%
      \unskip%
    }%
  \fi%
  \ifvoid\footins\else%
    \setbox\Zero\vbox{%
      \boxmaxdepth\maxdepth%
      \unvbox\Zero%
      \vskip\skip\footins%
      \footnoterule%
      \unvbox\footins%
    }%
  \fi%
  \setbox\Zero\vbox to\vsize{%
    \dimen\Zero\dp\Zero%
    \unvbox\Zero%
    \vskip -\dimen\Zero%
    \@ragged@bottom%
  }%
  \box\Zero%
}%
\def\makepageF{%
  % Clear the page box (it should have no content):
  {\setbox\Zero\box\Cclv}%
  \setbox\Zero\vbox{%
    \def\@reset@counta{\counta 25\relax}%
    \@reset@counta%
    \boxmaxdepth\maxdepth%
    \offinterlineskip%
    \global\setbox\One\vbox{}%
    \unvbox\topins%
    \unvbox\botins%
    \loop%
      \unskip\unpenalty%
      \setbox\Zero\lastbox%
      \ifvoid\Zero\advance\counta\MinusOne\fi%
      \ifvbox\Zero%
        \@reset@counta%
        \global\setbox\One\vbox{\box\Zero\vskip\fpsep\unvbox\One}%
      \fi%
      \ifnum\counta>\Zero%
    \repeat%
  }%
  \dimen\One\ht\One%
  \advance\dimen\One\dp\One%
  \ifvoid\footins%
    % Only floats:
    \ifdim\dimen\One>.5\topskip%
      \setbox\Zero\vbox to\vsize{%
        \vskip\fptop%
        \unvbox\One%
        \unskip%
        \vskip\fpbot%
      }%
    \else%
      \setbox\Zero\vbox to\ZeroUnit{}%
    \fi%
  \else%
    \setbox\One\vbox{%
      \boxmaxdepth\maxdepth%
      \ifdim\dimen\One>.5\topskip%
        % Footnotes goes after floats:
        \unvbox\One%
        \unskip%
        \vskip\skip\footins%
        \footnoterule%
      \else%
        % Footnotes goes on the separate page:
        \setbox\Zero\vtop{\unvcopy\footins}%
        \skip\Zero\topskip%
        \advance\skip\Zero-\ht\Zero%
        \ifdim\skip\Zero>\ZeroUnit\vskip\skip\Zero\fi% \topskip
      \fi%
      \unvbox\footins%
    }%
    \setbox\Zero\vbox to\vsize{%
      \dimen\One\dp\One%
      \unvbox\One%
      \vskip -\dimen\One%
      \@ragged@bottom%
    }%
  \fi%
  \ifdim\ht\Zero>\ZeroUnit\box\Zero\fi%
}%
\def\makefootline{%
  \baselineskip\footskip \lineskiplimit\ZeroUnit%
  \hbox to\hsize{\@the@foot}%
}%

% Full page:
\def\fullpage{%
  \hoffset -1in%
  \voffset -1in%
  \hsize\paperwidth%
  \vsize\paperheight%
  \topskip\ZeroUnit%
  \parindent\ZeroUnit%
  \offinterlineskip%
  \parskip\ZeroUnit%
  \everypar{{\setbox\Zero\lastbox}}%
  \output{\shipout\box\Cclv \advancepageno}%
}%
