\def\@setcmap{@setcmap@}%
\def\setcmap #1#2{{%
  \tolower\@tempa{\@setcmap #1}%
  \applycw\ifx\@tempa\relax%
    \tolower\@tempb{macros/cmaps/#1.cmap}%
    \testfile\@tempb%
    \iffile%
      \@pdfobj stream file {\@tempb}\relax%
      \namexdef\@tempa ##1{%
        \@pdfnobuiltintounicode ##1\relax%
        \@pdffontattr ##1 {/ToUnicode \@pdflastobj}\relax%
      }%
    \fi%
  \fi%
  \applyw\@tempa #2%
}}%
\def\cmap #1\font#2{\def\next{\setcmap{#1}#2}\afterassignment\next\font#2}%
\def\@process@deferred@cmaps{%
  \@deferred@cmaps%
  \let\@deferred@cmaps\empty%
}%

% Implement placing anchors into PDF:
\ifx\pdfdest\undefined\else%
  \def\@pdfdest@offset{.8em}%
  \def\@place@anchor{%
    \leavevmode%
    \vbox to \ZeroUnit{%
      \measuretexty\dimena{X}%
      \advance \dimena \@pdfdest@offset%
      \kern -\dimena%
      \pdfdest name {\@anchor@name \@anchor@num} xyz\relax%
      \vss%
    }%
  }%
\fi%

% Implement hyper links:
\ifx\pdfstartlink\undefined\else\ifx\pdfliteral\undefined\else%
  \def\@crosslink #1#2{%
    \applycw\ifx{#1}\relax%
      \warning{Undefined anchor `#1`.}%
    \else%
      \leavevmode%
      \pdfstartlink
        height .9em depth .3em
        attr {/Border [0 0 0] /H /N}
        goto name {#1}%
        \pdfliteral{0 0 1 rg}%
        #2%
        \pdfliteral{0 g}%
      \pdfendlink%
    \fi%
  }%
\fi\fi%

% Emit PDF outlines:
\newif\if@count@siblings \@count@siblingsfalse%
\def\@pdfoutline #1#2#3{%
  \if@count@siblings%
    \advance \counta \One%
  \else%
    \pdfoutline
      goto name {\@anchor@name #1} \ifnum#3>\Zero count \the#3\space\fi {#2}%
      \relax%
  \fi%
}%
\def\@outlines@flush{%
  \counta \Zero%
  \@count@siblingstrue%
  \@tempb%
  \@count@siblingsfalse%
  \ifx\@tempa\empty\else%
    \@tempa\counta%
    \let\@tempa\empty%
  \fi%
  \@tempb%
  \let\@tempb\empty%
}%
\def\@outlines@handle@part #1#2#3{%
  \@outlines@flush%
  \@pdfoutline{#3}{\if$#1$\else\partname\space#1 \fi#2}{0}%
}%
\def\@outlines@handle@chapter #1#2#3{%
  \@outlines@flush%
  \def\@tempa{\@pdfoutline{#3}{\if$#1$\else#1 \fi#2}}%
}%
\def\@outlines@handle@section #1#2#3{%
  \addtoks\@tempb{\@pdfoutline{#3}{\if$#1$\else#1 \fi#2}{0}}%
}%
\def\@emit@outlines{{%
  \let\@tempa\empty%
  \let\@tempb\empty%
  \def\tocline ##1##2##3##4##5{\applyw{@outlines@handle@##1}{##2}{##3}{##5}}%
  \@toclines%
  \@outlines@flush%
}}%
\ifx\pdfoutline\undefined\else%
  \addtoks\atdocend{\@emit@outlines}%
\fi%

\def\@emit@catalog{%
  \pdfcatalog {
    /PageMode \ifx\@toclines\empty /UseThumbs \else /UseOutlines \fi
    /PageLayout /OneColumn
    \ifx\@page@labels\empty\else
      /PageLabels << /Nums [ \@page@labels\space ] >>
    \fi
  }
  openaction goto page 1 {/FitH}\relax%
}%
